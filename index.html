using System.Collections; using UnityEngine; using UnityEngine.Networking; using System.Collections.Generic; [System.Serializable] public class CommentData { public string name; public string message; public string date; } [System.Serializable] public class CommentWrapper { public List data; } public class SheetDBManager : MonoBehaviour { // ⚠️ 換成你的 SheetDB API 網址 private string apiUrl = "YOUR_SHEETDB_API_URL"; void Start() { StartCoroutine(GetMessages()); } // 1. 讀取留言 (GET) IEnumerator GetMessages() { using (UnityWebRequest www = UnityWebRequest.Get(apiUrl)) { yield return www.SendWebRequest(); if (www.result == UnityWebRequest.Result.Success) { string jsonResult = www.downloadHandler.text; // SheetDB 回傳的通常是陣列，可以用 JsonUtility 處理 // (註：如果回傳是頂層陣列，Unity 內建 JsonUtility 需包一層 Wrapper 處理) Debug.Log("成功取得留言: " + jsonResult); } else { Debug.LogError("讀取失敗: " + www.error); } } } // 2. 發送留言 (POST) public void SendMessageToSheet(string playerName, string playerMessage) { StartCoroutine(PostCoroutine(playerName, playerMessage)); } IEnumerator PostCoroutine(string name, string message) { // 建立要上傳的資料結構 CommentData newComment = new CommentData { name = name, message = message, date = System.DateTime.Now.ToString("yyyy-MM-dd HH:mm") }; // SheetDB POST 格式要求需要包在 {"data": [...]} 裡面 string jsonBody = "{\"data\":[" + JsonUtility.ToJson(newComment) + "]}"; using (UnityWebRequest www = new UnityWebRequest(apiUrl, "POST")) { byte[] bodyRaw = System.Text.Encoding.UTF8.GetBytes(jsonBody); www.uploadHandler = new UploadHandlerRaw(bodyRaw); www.downloadHandler = new DownloadHandlerBuffer(); www.SetRequestHeader("Content-Type", "application/json"); yield return www.SendWebRequest(); if (www.result == UnityWebRequest.Result.Success) { Debug.Log("留言成功送出！"); StartCoroutine(GetMessages()); // 重新整理列表 } else { Debug.LogError("送出失敗: " + www.error); } } } }
